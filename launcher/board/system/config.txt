# Carrier board: append to /boot/firmware/config.txt (under [all]).
# The board's ID EEPROM can carry the same overlays once programmed.

dtparam=i2c_arm=on
dtparam=audio=off

# PI_HOLD (GPIO26) high as early as the firmware can, then held by
# gpio-poweroff until the kernel halts; the board cuts power ~2 s later.
gpio=26=op,dh
dtoverlay=gpio-poweroff,gpiopin=26,active_low=1

# DS3231 RTC on I2C-1 (0x68): the clock survives power off
dtoverlay=i2c-rtc,ds3231

# Audio: Pi I2S as clock consumer, 48 kHz stereo (see carrier-audio-overlay.dts)
dtoverlay=carrier-audio

# Fan PWM on GPIO12/13 (pwmchip0 channels 0/1)
dtoverlay=pwm-2chan,pin=12,func=4,pin2=13,func2=4
